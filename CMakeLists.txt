cmake_minimum_required(VERSION 3.16)
project(game_edge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(game_edge_frame src/frame.cpp)
target_include_directories(game_edge_frame PUBLIC include)
add_executable(frame_test tests/frame_test.cpp)
target_link_libraries(frame_test PRIVATE game_edge_frame)
add_library(game_edge_net src/net.cpp src/poller.cpp)
target_include_directories(game_edge_net PUBLIC include)
if(WIN32)
  target_link_libraries(game_edge_net PUBLIC ws2_32)
endif()
add_executable(net_test tests/net_test.cpp)
target_link_libraries(net_test PRIVATE game_edge_net)
add_executable(game_edge src/gateway.cpp src/main.cpp)
target_link_libraries(game_edge PRIVATE game_edge_frame game_edge_net)
enable_testing()
add_test(NAME frame_test COMMAND frame_test)
add_test(NAME net_test COMMAND net_test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME gateway_test COMMAND ${CMAKE_COMMAND} -E env "GAME_EDGE_BIN=$<TARGET_FILE:game_edge>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_gateway.py -v)
  add_test(NAME bench_test COMMAND ${CMAKE_COMMAND} -E env "GAME_EDGE_BIN=$<TARGET_FILE:game_edge>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_bench.py -v)
  add_test(NAME experiment_test COMMAND ${CMAKE_COMMAND} -E env "GAME_EDGE_BIN=$<TARGET_FILE:game_edge>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_experiment.py -v)
  add_test(NAME routing_test COMMAND ${CMAKE_COMMAND} -E env "GAME_EDGE_BIN=$<TARGET_FILE:game_edge>" ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p test_routing.py -v)
  set_tests_properties(gateway_test bench_test experiment_test routing_test PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
